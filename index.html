const API_BASE = "https://rifhskboy-github-io.rifhskstore.workers.dev"; async function loadLiveOrders() { try { const response = await fetch(`${API_BASE}/api/orders/recent`); const data = await response.json(); if (!data.success) return; const container = document.getElementById("live-orders"); if (!container) return; container.replaceChildren(); data.orders.forEach(order => { const item = document.createElement("div"); item.className = "live-order-item"; const service = document.createElement("strong"); service.textContent = order.service; const detail = document.createElement("p"); detail.textContent = `${order.quantity} match · Rp${Number(order.total).toLocaleString("id-ID")}`; const status = document.createElement("small"); status.textContent = order.status; item.append(service, detail, status); container.appendChild(item); }); } catch (error) { console.error("Gagal memuat Live Order:", error); } } loadLiveOrders(); setInterval(loadLiveOrders, 15000);
